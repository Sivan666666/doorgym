# Door DP 常用命令
#
# 默认从仓库根目录运行：
#   cd /home/sivan/whole_body/visual_whole_body
#
# 当前主线数据格式：
# - 仿真/control 仍是 50Hz，也就是 dt=0.02s。
# - raw 数据按 25Hz 保存：state/action/camera 都是 25Hz。
# - vision 使用 depth_only：只保存两张图
#     observation.images.wrist_masked_depth
#     observation.images.front_masked_depth
#   不保存/不训练 handle mask。
# - state 是 10D 物理当前状态：
#     [vx, yaw_rate, ee_x, ee_y, ee_z, ee_qx, ee_qy, ee_qz, ee_qw, gripper]
# - raw 转 LeRobot 时 state/action 不做 Door 自定义归一化；LeRobot train 自己根据 dataset stats 做 normalization。
# - scripted ikpush 默认已缩短：initial_hold=150, initial_hold_move=100, grasp_hold=5, gripper_close=100。
# - 门两侧墙体默认开启；如需关闭，在底层 float_ik 参数里加 --no_door_side_walls。


PROJECT_ROOT=/home/sivan/whole_body/visual_whole_body

REPO_ID=local/door_pi05_state10_depthonly_wall_40each

ACT_RUN=leroact_door_pi05_state10_depthonly_wall_chunk100_exec50_bs16_no_imagenet_0604_2117
ACT_CKPT=$PROJECT_ROOT/high-level/dp/logs/lerobot-train/$ACT_RUN/checkpoints/100000

    --rl_device cuda:0 \
    --sim_device cuda:0 \
    --headless \
    --depth_only \
    -- \
    --door_cfg high-level/experiments/isaacgym/b1z1_opendoor.yaml \
    --no_show_seg

# 单次录制调试：只跑一次 16 env。
python high-level/dp/record/record_door_dp_dataset_pi05_state10.py \
    --mode ikpush \

    --raw_root "$RAW_ROOT"_debug \

    --sim_device cuda:0 \
    --headless \
    --depth_only \
    -- \
    --door_cfg high-level/experiments/isaacgym/b1z1_opendoor.yaml \

python high-level/dp/record/replay_door_dp_raw_in_isaacgym.py \

## 4. raw 转 LeRobotDataset

# 转成 LeRobot v3 video 格式。这里显式指定 --fps 25 和 --depth_only。
# near_zero_rate_eps 只清理 yaw/yaw_rate 一类接近 0 的数值抖动，不做归一化。
conda run -n b1z1_lerobot python high-level/dp/convert_door_raw_to_lerobot.py \
    --raw_root "$RAW_ROOT" \
    --root "$LEROBOT_ROOT" \
    --repo_id "$REPO_ID" \
    --overwrite \
    --fps 25 \
    --depth_only \
    --image_storage video \
    --video_codec h264 \
    --num_workers 4 \
    --state_preprocess none \
    --action_preprocess none \
    --near_zero_rate_eps 1e-5

# 可视化检查 LeRobotDataset。
conda run -n b1z1_lerobot python -m lerobot.scripts.visualize_dataset \
    --repo-id "$REPO_ID" \
    --root "$DATASET_ROOT" \
    --episode-index 0 \
    --mode local


## 5. 训练 policy

# 推荐用官方 LeRobot train.py，完整命令见 high-level/dp/command_lerobot.txt。
# 关键训练约定：
# - ACT: chunk_size=100, n_action_steps=50。25Hz 下预测 4s，执行 2s。
# - DP: horizon=40, n_action_steps=20。25Hz 下预测 1.6s，执行 0.8s。
# - 所有 LeRobot train 命令都加 --dataset.use_imagenet_stats=false。

# 自己 train 文件夹里的 ACT 训练脚本，使用同一份 LeRobot 数据集。
conda run -n b1z1_lerobot python high-level/dp/train/train_door_act.py \
    --root "$LEROBOT_ROOT" \
    --repo_id "$REPO_ID" \
    --run_name act_depthonly_wall_chunk100_exec50_bs16 \
    --steps 100000 \
    --batch_size 16 \
    --num_workers 4 \
    --device cuda:0 \
    --chunk_size 100 \
    --action_horizon 50 \
    --wandb
