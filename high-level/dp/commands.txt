Phase B: raw .npz 转 LeRobotDataset

Phase D: play / eval success / expert obs 对比

## 5. 训练 policy

# DP 训练。预测 40 步，执行 20 步；play/eval 推理使用 DDPM 10 denoising steps。
conda run -n door_dp python high-level/dp/train/train_door_dp.py \
    --root high-level/data/lerobot \
    --repo_id local/4door_ikpush \
    --run_name lerodp_4door_ikpush_h40_exec20 \
    --steps 100000 \
    --batch_size 16 \
    --num_workers 4 \
    --device cuda:0 \
    --obs_horizon 16 \
    --pred_horizon 40 \
    --action_horizon 20 \
    --num_diffusion_iters 100 \
    --num_inference_steps 10 \
    --noise_scheduler_type DDPM \
    --clip_sample_range 7.0 \
    --wandb

# ACT 训练。预测 100 步，执行 50 步。
conda run -n door_dp python high-level/dp/train/train_door_act.py \
    --root high-level/data/lerobot \
    --repo_id local/4door_ikpush \
    --run_name act_4door_ikpush_chunk100_exec50_bs16 \
    --steps 100000 \
    --batch_size 16 \
    --num_workers 4 \
    --device cuda:0 \
    --chunk_size 100 \
    --action_horizon 50 \
    --wandb

# pi0.5 训练，需要先准备 pretrained_path。预测 50 步，执行 50 步。
conda run -n door_dp python high-level/dp/train/train_door_pi05.py \
    --root high-level/data/lerobot \
    --repo_id local/4door_ikpush \
    --run_name pi05_4door_ikpush \
    --steps 100000 \
    --batch_size 1 \
    --num_workers 0 \
    --device cuda:0 \
    --chunk_size 50 \
    --action_horizon 50 \
    --num_inference_steps 10 \
    --pretrained_path <pi05_checkpoint_or_repo>


## 6. Play trained policy

# DP / ACT / pi0.5 统一入口，会自动根据 checkpoint metadata 识别 policy 类型。
python high-level/dp/play/play_door_policy.py \
    --mode ikpush \
    --checkpoint high-level/dp/logs/door-auto-wrapped/lerodp_4door_ikpush_h40_exec20_bs16_no_imagenet/100000/model_latest.pt \
    --num_envs 4 \
    --steps 2500 \
    --rl_device cuda:0 \
    --sim_device cuda:0 \
    --dp_inference_steps 10 \
    --dp_noise_scheduler_type DDPM \
    --dp_action_horizon 20 \
    -- \
    --door_cfg high-level/experiments/isaacgym/b1z1_opendoor.yaml

# ACT checkpoint 示例。ACT 预测 100 步，执行 50 步。
# dp_inference_steps 对 ACT 没作用，保留在命令里方便切回 DP。
python high-level/dp/play/play_door_policy.py \
    --mode ikpush \
    --checkpoint high-level/dp/logs/door-act/act_4door_ikpush_chunk100_exec50_bs16/checkpoints/model_latest.pt \
    --num_envs 4 \
    --steps 2500 \
    --rl_device cuda:0 \
    --sim_device cuda:0 \
    --dp_inference_steps 10 \
    --dp_noise_scheduler_type DDPM \
    --dp_action_horizon 50 \
    -- \
    --door_cfg high-level/experiments/isaacgym/b1z1_opendoor.yaml

# 从 raw expert 某一帧 warmstart，再闭环 play policy。
python high-level/dp/play/play_door_policy.py \
    --mode ikpush \
