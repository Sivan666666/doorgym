# Door pi0.5 / pi0.5 evo commands
#
# This file is only for pi0.5-family policies. General ACT/DP commands live in:
#   high-level/dp/command_lerobot.txt
#
# Current dataset:
# - LeRobot repo_id: local/door_pi05_state10_depthonly_wall_40each
# - 25Hz depth_only data over 50Hz Isaac Gym simulation.
# - policy.chunk_size=100 predicts 4.0s of actions.
# - policy.n_action_steps=50 executes 2.0s before the next chunk is requested.

set -o pipefail

PROJECT_ROOT=/home/sivan/whole_body/visual_whole_body
DATASET_REPO_ID=local/door_pi05_state10_depthonly_wall_40each
DATASET_ROOT=$PROJECT_ROOT/high-level/data/lerobot/local/door_pi05_state10_depthonly_wall_40each
OUTPUT_ROOT=$PROJECT_ROOT/high-level/dp/logs/lerobot-train

cd "$PROJECT_ROOT"


## 1. LeRobot pi0.5 baseline
#
# Environment:
#   conda activate b1z1_lerobot
# Source:
#   high-level/lerobot

export PYTHONPATH="$PROJECT_ROOT/high-level/lerobot/src:${PYTHONPATH:-}"

PI05_RUN_NAME=pi05_door_pi05_state10_depthonly_wall_chunk100_exec50_no_imagenet_$(date +%m%d_%H%M)

CUDA_VISIBLE_DEVICES=0 accelerate launch \
    --num-processes=1 \
    --mixed_precision=bf16 \
    "$(which lerobot-train)" \
    --dataset.root="$DATASET_ROOT" \
    --dataset.repo_id="$DATASET_REPO_ID" \
    --dataset.video_backend=torchcodec \
    --dataset.use_imagenet_stats=false \
    --policy.type=pi05 \
    --policy.device=cuda \
    --policy.pretrained_path=lerobot/pi05_base \
    --policy.dtype=bfloat16 \
    --policy.push_to_hub=false \
    --policy.gradient_checkpointing=true \
    --policy.chunk_size=100 \
    --policy.n_action_steps=50 \
    --policy.num_inference_steps=10 \
    --batch_size=16 \
    --steps=30000 \
    --num_workers=4 \
    --save_freq=5000 \
    --log_freq=50 \
    --job_name="$PI05_RUN_NAME" \
    --output_dir="$OUTPUT_ROOT/$PI05_RUN_NAME" \
    --wandb.enable=true \
    --wandb.project=door-pi05 \
    --wandb.disable_artifact=true


## 2. HWK LeRobot pi0.5 evo
#
# Environment:
#   conda activate b1z1_hwk_lerobot
# Source:
#   high-level/hwk/lerobot
# Note:
#   pi05_evo is provided by the HWK source tree, so HWK must be first in PYTHONPATH.

export PYTHONPATH="$PROJECT_ROOT/high-level/hwk/lerobot/src:$PROJECT_ROOT/high-level/lerobot/src:${PYTHONPATH:-}"

PI05_EVO_RUN_NAME=pi05_evo_door_pi05_state10_depthonly_wall_chunk100_exec50_no_imagenet_$(date +%m%d_%H%M)

CUDA_VISIBLE_DEVICES=0 accelerate launch \
    --num-processes=1 \
    --mixed_precision=bf16 \
    "$(which lerobot-train)" \
    --dataset.root="$DATASET_ROOT" \
    --dataset.repo_id="$DATASET_REPO_ID" \
    --dataset.video_backend=torchcodec \
    --dataset.use_imagenet_stats=false \
    --policy.type=pi05_evo \
    --policy.device=cuda \
    --policy.pretrained_path=lerobot/pi05_base \
    --policy.dtype=bfloat16 \
    --policy.push_to_hub=false \
    --policy.gradient_checkpointing=true \
    --policy.chunk_size=100 \
    --policy.n_action_steps=50 \
    --policy.num_inference_steps=10 \
    --batch_size=16 \
    --steps=30000 \
    --num_workers=4 \
    --save_freq=5000 \
    --log_freq=50 \
    --job_name="$PI05_EVO_RUN_NAME" \
    --output_dir="$OUTPUT_ROOT/$PI05_EVO_RUN_NAME" \
    --wandb.enable=true \
    --wandb.project=door-pi05 \
    --wandb.disable_artifact=true


## 3. Play pi0.5-family checkpoint
#
# Runtime environment:
#   conda activate b1z1
# The checkpoint can be an official LeRobot checkpoint directory. play_door_policy.py
# will auto-wrap it for Door runtime.

PI05_CKPT="$OUTPUT_ROOT/$PI05_RUN_NAME/checkpoints/30000"

python "$PROJECT_ROOT/high-level/dp/play/play_door_policy.py" \
    --mode ikpush \
    --checkpoint "$PI05_CKPT" \
    --num_envs 16 \
    --steps 2500 \
    --rl_device cuda:0 \
    --sim_device cuda:0 \
    --depth_only \
    --dp_action_horizon 50 \
    --camera_display_scale 5 \
    -- \
    --door_cfg "$PROJECT_ROOT/high-level/experiments/isaacgym/b1z1_opendoor.yaml" \
    --pass_open_angle_deg 80 \
    --dp_fps 25 \
    --seed 123456
