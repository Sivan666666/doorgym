# Official LeRobot training / play commands for Door DP / ACT / pi0.5.
#
# Run from anywhere. Change RUN_NAME if you start a fresh training run.
# Official LeRobot train.py refuses to reuse an existing output_dir.
#
# Current horizon convention:
# - Diffusion Policy: predict 40 control steps, execute 20 control steps.
#   At 50 Hz this is predict 0.8s, execute 0.4s.
# - ACT: predict 100 control steps, execute 50 control steps.
#   At 50 Hz this is predict 2.0s, execute 1.0s.
# - pi0.5 / pi0.5 evo: predict 50 control steps, execute 50 control steps.
#   num_inference_steps is diffusion/flow denoising steps, not robot execution steps.
#
# Important:
# - All official LeRobot train commands use --dataset.use_imagenet_stats=false.
#   Door images are depth/mask, not RGB ImageNet images.
# - push_to_hub is disabled for all policies.
# - wandb is enabled, artifact upload is disabled.

set -o pipefail

PROJECT_ROOT=/home/sivan/whole_body/visual_whole_body
DATASET_REPO_ID=local/4door_ikpush
DATASET_ROOT=$PROJECT_ROOT/high-level/data/lerobot/local/4door_ikpush
OUTPUT_ROOT=$PROJECT_ROOT/high-level/dp/logs/lerobot-train

cd "$PROJECT_ROOT"


## 1. Diffusion Policy, official LeRobot train.py
#
# Predict 40 steps, execute 20 steps.
# drop_n_last_frames = horizon - n_action_steps - n_obs_steps + 1 = 40 - 20 - 16 + 1 = 5.

export PYTHONPATH="$PROJECT_ROOT/high-level/lerobot/src:${PYTHONPATH:-}"

DP_RUN_NAME=lerodp_4door_ikpush_h40_exec20_bs16_no_imagenet

CUDA_VISIBLE_DEVICES=0 accelerate launch \
    --num-processes=1 \
    --mixed_precision=no \
    "$(which lerobot-train)" \
    --dataset.root="$DATASET_ROOT" \
    --dataset.repo_id="$DATASET_REPO_ID" \
    --dataset.video_backend=torchcodec \
    --dataset.use_imagenet_stats=false \
    --policy.type=diffusion \
    --policy.device=cuda \
    --policy.push_to_hub=false \
    --policy.n_obs_steps=16 \
    --policy.horizon=40 \
    --policy.n_action_steps=20 \
    --policy.drop_n_last_frames=5 \
    --policy.noise_scheduler_type=DDPM \
    --policy.num_train_timesteps=100 \
    --policy.num_inference_steps=10 \
    --policy.clip_sample_range=7.0 \
    --batch_size=16 \
    --steps=100000 \
    --num_workers=4 \
    --save_freq=1000 \
    --log_freq=10 \
    --job_name="$DP_RUN_NAME" \
    --output_dir="$OUTPUT_ROOT/$DP_RUN_NAME" \
    --wandb.enable=true \
    --wandb.project=door-dp \
    --wandb.disable_artifact=true

## 5. Play official LeRobot Diffusion checkpoint in Isaac Gym
#
# Runtime environment:
#   conda activate b1z1
#
# The first run auto-wraps the official LeRobot checkpoint to Door format under
# high-level/dp/logs/door-auto-wrapped/.
# DP inference uses DDPM with 10 denoising steps. Robot executes 20 control steps per policy query.

export DOOR_DP_LEROBOT_CONDA_ENV=b1z1_lerobot

python "$PROJECT_ROOT/high-level/dp/play/play_door_policy.py" \
    --mode ikpush \
    --checkpoint "$OUTPUT_ROOT/$DP_RUN_NAME/checkpoints/last" \
    --num_envs 16 \
    --steps 2500 \
    --rl_device cuda:0 \
    --sim_device cuda:0 \
    --dp_action_horizon 20 \
    --dp_inference_steps 10 \
    --dp_noise_scheduler_type DDPM \
    --camera_display_scale 5 \
    -- \
    --door_cfg "$PROJECT_ROOT/high-level/experiments/isaacgym/b1z1_opendoor.yaml" \
    --pass_open_angle_deg 80 \
    --seed 123456
