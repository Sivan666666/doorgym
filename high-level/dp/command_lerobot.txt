# Official LeRobot train / play / eval commands for Door policies.
#
# Run from anywhere. These commands target the current main dataset:
#   local/door_pi05_state10_depthonly_wall_40each
#
# Dataset/runtime convention:
# - Isaac Gym simulation/control: 50Hz, dt=0.02s.
# - Dataset frames: 25Hz. One dataset/action step = 0.04s.
# - Vision mode: depth_only, two image keys only:
#     observation.images.wrist_masked_depth
#     observation.images.front_masked_depth
# - State/action in LeRobot files are physical values; LeRobot train applies its own stats normalization.
# - All train commands use --dataset.use_imagenet_stats=false because depth images are not ImageNet RGB.
# - push_to_hub is disabled. wandb is enabled, artifact upload is disabled.

set -o pipefail

PROJECT_ROOT=/home/sivan/whole_body/visual_whole_body
DATASET_REPO_ID=local/door_pi05_state10_depthonly_wall_40each
DATASET_ROOT=$PROJECT_ROOT/high-level/data/lerobot/local/door_pi05_state10_depthonly_wall_40each
RAW_ROOT=$PROJECT_ROOT/high-level/data/door_dp_raw/4door_ikpush_pi05_state10_depthonly_wall_40each
OUTPUT_ROOT=$PROJECT_ROOT/high-level/dp/logs/lerobot-train

cd "$PROJECT_ROOT"
export PYTHONPATH="$PROJECT_ROOT/high-level/lerobot/src:${PYTHONPATH:-}"


## 1. Diffusion Policy, official LeRobot train.py
#
# Horizon:
# - predict 40 action steps = 1.6s at 25Hz.
# - execute 20 action steps = 0.8s at 25Hz.
# - drop_n_last_frames = horizon - n_action_steps - n_obs_steps + 1 = 40 - 20 - 16 + 1 = 5.

DP_RUN_NAME=lerodp_door_pi05_state10_depthonly_wall_h40_exec20_bs16_no_imagenet

CUDA_VISIBLE_DEVICES=0 accelerate launch \
    --num-processes=1 \
    --mixed_precision=no \
    "$(which lerobot-train)" \
    --dataset.root="$DATASET_ROOT" \
    --dataset.repo_id="$DATASET_REPO_ID" \
    --dataset.video_backend=torchcodec \
    --dataset.use_imagenet_stats=false \
    --policy.type=diffusion \
    --policy.device=cuda \
    --policy.push_to_hub=false \
    --policy.n_obs_steps=16 \
    --policy.horizon=40 \
    --policy.n_action_steps=20 \
    --policy.drop_n_last_frames=5 \
    --policy.noise_scheduler_type=DDPM \
    --policy.num_train_timesteps=100 \
    --policy.num_inference_steps=10 \
    --policy.clip_sample_range=7.0 \
    --batch_size=16 \
    --steps=100000 \
    --num_workers=4 \
    --save_freq=10000 \
    --log_freq=10 \
    --job_name="$DP_RUN_NAME" \
    --output_dir="$OUTPUT_ROOT/$DP_RUN_NAME" \
    --wandb.enable=true \
    --wandb.project=door-dp \
    --wandb.disable_artifact=true

# Optional ACT front/wrist camera input gating (disabled by default).
# Add these flags to the ACT command above to enable it:
#
#   --policy.camera_input_gating=true \
#   --policy.camera_input_gating_hidden_dim=128 \
#   --policy.camera_input_gating_temperature=1.0 \

# The final gating layer is zero-initialized and uses 2*softmax, so training

## 5. Play official LeRobot ACT checkpoint in Isaac Gym
#
# Runtime environment:
#   conda activate b1z1
# The first run auto-wraps the official LeRobot checkpoint to Door format.
# --dp_fps 25 is forwarded to float_ik and makes each action command last 0.04s.

export DOOR_DP_LEROBOT_CONDA_ENV=b1z1_lerobot

ACT_CKPT="$OUTPUT_ROOT/$ACT_RUN_NAME/checkpoints/100000"

python "$PROJECT_ROOT/high-level/dp/play/play_door_policy.py" \
    --mode ikpush \
    --checkpoint "$ACT_CKPT" \
    --num_envs 16 \
    --steps 2500 \
    --rl_device cuda:0 \
    --sim_device cuda:0 \
    --depth_only \
    --dp_action_horizon 50 \
    --camera_display_scale 1 \
    -- \
    --door_cfg "$PROJECT_ROOT/high-level/experiments/isaacgym/b1z1_opendoor.yaml" \
    --pass_open_angle_deg 80 \
    --dp_fps 25 \
    --seed 123456


## 6. Expert-observation eval

    --raw_episode "$RAW_ROOT/episode_000000.npz" \
    --checkpoint "$ACT_CKPT" \

    --action_horizon 50 \

    --end 1200 \
    --stride 50 \
